# ============================================================================
# UPSTREAM - Health Server (porta 8090)
# ============================================================================
# Health server local que agrega status de todos os backends
upstream health_server {
    server localhost:8090;
    keepalive 2;
    keepalive_timeout 60s;
    keepalive_requests 100;
}

# ============================================================================
# CONFIGURAÇÕES COMPARTILHADAS - Map para variáveis
# ============================================================================
# Define tipo de host baseado em variável de ambiente
map $server_name $host_type {
    default "load-balancer";
}

# ============================================================================
# TROUBLESHOOTING
# ============================================================================
#
# Problema: "connect() failed (111: Connection refused)"
# Solução: Health-server não está rodando
#          systemctl --user start health-server
#          curl http://localhost:8090/health
#
# Problema: "upstream 'health_server' not found"
# Solução: Adicionar bloco upstream no contexto http
#
# Problema: 502 Bad Gateway em /health
# Solução: Verificar logs do health-server
#          journalctl --user -u health-server -n 50
#
# Problema: 404 Not Found
# Solução: Locations devem estar dentro de server { ... }
#
