# Descrição:

#   NÃO inclui health check agregado (sem health-server).

# Diferenças do Load Balancer:
#   ❌ Não tem upstream health_server

#     add_header Cache-Control "no-cache, no-store, must-revalidate";
# }

# 2. [ALTERNATIVA] Health com verificação de container/backend
# ----------------------------------------------------------------------------
# Este endpoint faz proxy para um dos containers backend

#             Este endpoint retornará erro se o container estiver down
#
# USO: Descomente e adapte para seu container principal
#
# location /health {
#     # Proxy para container backend (exemplo: API principal)
#     proxy_pass http://MainApi:5000/health;
#
#     # Timeout curto (falha rápida se container estiver down)
#     proxy_connect_timeout 2s;
#     proxy_send_timeout 2s;
#     proxy_read_timeout 2s;
#
#     # Headers
#     proxy_set_header Host $host;
#     proxy_set_header X-Real-IP $remote_addr;
#     proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
#
#     # Sem cache
#     add_header Cache-Control "no-cache, no-store, must-revalidate";
#     access_log off;
#
#     # Fallback: retorna 200 mesmo se container não tiver /health
#     proxy_intercept_errors on;

# INSTALAÇÃO NO WEBSERVER
# ============================================================================
#
# 1. Copiar este arquivo:
#    sudo cp /workspaces/pipelines-store/nginx/Common/004health-endpoints-web.conf \
#            /etc/nginx/conf.d/004health-endpoints-web.conf
#
# 2. Editar nginx.conf ou proxy-webserver-prd.conf:
#    - Copiar locations necessárias para dentro do server { ... }
#    - Escolher entre /nginx-health (simples) ou /health (com backend check)
#
# 3. Testar sintaxe NGINX:
#    sudo nginx -t
#
# 4. Recarregar NGINX:
#    sudo systemctl reload nginx
#    # OU
#    podman exec nginx nginx -s reload
#
# 5. Validar endpoints:
#    curl http://webserver-01/nginx-health | jq
#    curl http://webserver-01/ping
#
# 6. Testar do Load Balancer:
#    curl http://webserver-01:80/nginx-health
#    curl http://webserver-02:80/nginx-health
#
# ============================================================================
# TROUBLESHOOTING
# ============================================================================
#
# Problema: 404 Not Found nos endpoints
# Solução: Locations devem estar dentro de server { ... }
#
# Problema: 502 Bad Gateway em /health (com backend check)
# Solução: Container backend está down ou não tem endpoint /health
#          Use location @health_fallback para degradar gracefully
#
# Problema: Load Balancer marca WebServer como down
# Solução: Verificar se /nginx-health ou /ping está respondendo 200
#          curl http://webserver-ip/nginx-health
#
# ============================================================================
# SEGURANÇA E RATE LIMITING
# ============================================================================
#
# 1. /nginx-status já está RESTRITO (allow/deny)
#
# 2. /nginx-health e /ping são públicos mas não expõem dados sensíveis
#
# 3. Rate limiting para endpoints públicos:
#
#    # No contexto http { ... }
#    limit_req_zone $binary_remote_addr zone=healthcheck_web:10m rate=20r/s;
#
#    # No contexto location
#    location /nginx-health {
#        limit_req zone=healthcheck_web burst=40 nodelay;
#        # ... resto da config ...
#    }
#
# 4. Restringir health checks apenas ao Load Balancer:
#
#    # Permitir apenas IP do Load Balancer
#    geo $lb_network {
#        default 0;
#        10.0.1.10/32 1;  # IP do Load Balancer
#        10.0.0.0/24 1;   # Rede do Load Balancer
#    }
#
#    location /nginx-health {
#        if ($lb_network = 0) {
#            return 403;
#        }
#        # ... resto da config ...
#    }
#
# ============================================================================
# INTEGRAÇÃO COM LOAD BALANCER
# ============================================================================
#
# No Load Balancer, configure health checks ativos para os WebServers:
#
# upstream webservers {
#     # WebServer 1
#     server 10.0.2.10:80 max_fails=3 fail_timeout=30s;
#     # WebServer 2
#     server 10.0.2.11:80 max_fails=3 fail_timeout=30s;
#
#     # Keepalive
#     keepalive 32;
# }
#
# # Health check passivo (NGINX Plus) ou use nginx_upstream_check_module
# # Para NGINX Open Source, use health-server.sh no Load Balancer
#
# Exemplo de health-server.sh verificando WebServers:
#
# BACKENDS="
# webserver1|http://10.0.2.10/nginx-health
# webserver2|http://10.0.2.11/nginx-health
# "
#
# ============================================================================
# MONITORAMENTO
# ============================================================================
#
# 1. Health check do Load Balancer:
#    - Load Balancer verifica /nginx-health ou /ping dos WebServers
#    - Se falhar, remove WebServer da rotação
#
# 2. Monitoramento externo:
#    - Use /nginx-health para status básico
#    - Use /health (com backend check) para validar stack completo
#    - Use /nginx-status para métricas do NGINX
#
# 3. Logs de acesso:
#    - Health checks têm access_log off para reduzir I/O
#    - Se precisar auditar, habilite logs com nível específico:
#
#      location /nginx-health {
#          access_log /var/log/nginx/health-access.log combined;
#          error_log /var/log/nginx/health-error.log warn;
#          # ... resto da config ...
#      }
#
# ============================================================================
# COMPARAÇÃO: Load Balancer vs WebServer
# ============================================================================
#
# ┌─────────────────────┬──────────────────────┬────────────────────────┐
# │ Endpoint            │ Load Balancer        │ WebServer              │
# ├─────────────────────┼──────────────────────┼────────────────────────┤
# │ /health             │ ✅ Agregado (8090)   │ ❌ ou ✅ Backend check │
# │ /nginx-health       │ ✅ Status LB         │ ✅ Status WebServer    │
# │ /ping               │ ✅ Ping LB           │ ✅ Ping WebServer      │
# │ /nginx-status       │ ✅ Métricas LB       │ ✅ Métricas WebServer  │
# │ health-server.sh    │ ✅ Requerido         │ ❌ Não necessário      │
# │ Upstream backend    │ ✅ Para apps         │ ✅ Para containers     │
# └─────────────────────┴──────────────────────┴────────────────────────┘
